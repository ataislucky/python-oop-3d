import math

class Bentuk:  # Kontrak dasar
    def hitung_luas(self):
        raise NotImplementedError("Wajib di-override!")

class Lingkaran(Bentuk):
    def __init__(self, r): self.r = r
    def hitung_luas(self): return math.pi * self.r ** 2

class Persegi(Bentuk):
    def __init__(self, sisi): self.sisi = sisi
    def hitung_luas(self): return self.sisi ** 2

class Segitiga(Bentuk):
    def __init__(self, a, t): self.a, self.t = a, t
    def hitung_luas(self): return 0.5 * self.a * self.t

class RobotGambar:  # TIDAK mewarisi Bentuk (Duck Typing)
    def __init__(self, p, l): self.p, self.l = p, l
    def hitung_luas(self): return self.p * self.l

# --- Fungsi Polimorfik Universal ---
def cetak_luas(daftar_objek):
    for item in daftar_objek:
        # Satu panggilan identik untuk semua tipe objek!
        print(f"{type(item).__name__}: {item.hitung_luas():.2f}")

daftar = [Lingkaran(7), Persegi(8), Segitiga(10, 6), RobotGambar(12, 5)]
cetak_luas(daftar)
