pewarisan_kendaraan.py
MRO: Mobil → Kendaraan → object 
class Kendaraan:  # Superclass / Parent
    def __init__(self, merk, tahun):
        self.merk = merk
        self.tahun = tahun

    def nyalakan_mesin(self):
        return f"Mesin {self.merk} menyala: Brumm!"

    def info(self):
        return f"{self.merk} ({self.tahun})"

class Mobil(Kendaraan):  # Subclass 1
    def __init__(self, merk, tahun, jumlah_pintu):
        super().__init__(merk, tahun)  # Panggil Parent
        self.jumlah_pintu = jumlah_pintu

    def info(self):  # Method Overriding
        dasar = super().info()
        return f"Mobil {dasar} • {self.jumlah_pintu} Pintu"

class Motor(Kendaraan):  # Subclass 2
    def __init__(self, merk, tahun, jenis_stang):
        super().__init__(merk, tahun)
        self.jenis_stang = jenis_stang

    def info(self):  # Method Overriding
        return f"Motor {self.merk} • Stang {self.jenis_stang}"

# --- Eksekusi & Pengujian ---
avanza = Mobil("Toyota", 2024, 4)
ninja = Motor("Kawasaki", 2025, "Jepit")
